cmake_minimum_required(VERSION 3.16)
project(memo_preview C)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH ${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(memo_preview preview.c ../../main/memo_ui.c ../../main/ui_pixel.c ../../main/memo_core.c ../../main/memo_font_16.c ../../main/memo_clock.c)
target_include_directories(memo_preview PRIVATE . ../../main)
target_link_libraries(memo_preview PRIVATE lvgl m)
